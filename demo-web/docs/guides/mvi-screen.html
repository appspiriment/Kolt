Build your first MVI Screen

Step-by-step tutorial on building a feature screen from scratch using Kolt MVI architecture and components.

Introduction

Kolt enforces a clean, unidirectional data flow (UDF) through a standard Model-View-Intent (MVI) pattern. In this guide, we will build a product details screen using real Kolt composables:

  • AppsPageScaffold for page structure and top bar navigation.
  • AsyncStateBox to switch layout states dynamically (loading, success, error).
  • AppsButton for interactive actions.
  • AppsBanner for system notifications and warnings.

1. Define the Contracts

Every screen starts by defining its contracts. Create a contract file containing your State, Intent, and Effect classes:

// ProductDetailsContract.kt
data class ProductDetailsState(
    val asyncState: AsyncState<Product> = AsyncState.Idle,
    val isFavorite: Boolean = false,
    val bannerMessage: String? = null
)

sealed interface ProductDetailsIntent {
    data class Load(val productId: String) : ProductDetailsIntent
    data object ToggleFavorite : ProductDetailsIntent
    data object ClearBanner : ProductDetailsIntent
}

sealed interface ProductDetailsEffect {
    data class ShowToast(val message: String) : ProductDetailsEffect
    data object NavigateBack : ProductDetailsEffect
}

2. Implement the ViewModel

Extend MviViewModel to manage state updates and side-effects. This keeps state mutation atomic and clean:

// ProductDetailsViewModel.kt
class ProductDetailsViewModel(
    private val getProductUseCase: GetProductUseCase,
    private val toggleFavoriteUseCase: ToggleFavoriteUseCase,
    private val dispatchers: DispatcherProvider
) : MviViewModel<ProductDetailsState, ProductDetailsIntent, ProductDetailsEffect>(
    ProductDetailsState()
) {
    override suspend fun onIntent(intent: ProductDetailsIntent) = when (intent) {
        is ProductDetailsIntent.Load -> {
            updateState { copy(asyncState = AsyncState.Loading) }
            val result = withContext(dispatchers.io) { getProductUseCase(intent.productId) }
            result.fold(
                onSuccess = { product ->
                    updateState { copy(asyncState = AsyncState.Success(product)) }
                },
                onFailure = { error ->
                    updateState { copy(asyncState = AsyncState.Error(error)) }
                }
            )
        }
        ProductDetailsIntent.ToggleFavorite -> {
            val currentState = state.value
            val currentProductState = currentState.asyncState
            if (currentProductState is AsyncState.Success) {
                val nextFavorite = !currentState.isFavorite
                updateState { copy(isFavorite = nextFavorite) }
                withContext(dispatchers.io) {
                    toggleFavoriteUseCase(currentProductState.data.id, nextFavorite)
                }
                sendEffect(ProductDetailsEffect.ShowToast(
                    if (nextFavorite) "Added to Favorites" else "Removed from Favorites"
                ))
            }
        }
        ProductDetailsIntent.ClearBanner -> {
            updateState { copy(bannerMessage = null) }
        }
    }
}

3. Create the Route Composable

The Route is the entry point that couples the ViewModel to your screen layout. It is the only place in the file that knows about the ViewModel class:

// ProductDetailsRoute.kt
@Composable
fun ProductDetailsRoute(
    productId: String,
    viewModel: ProductDetailsViewModel = hiltViewModel(),
    onBack: () -> Unit
) {
    val state by viewModel.state.collectAsStateWithLifecycle()

    // Trigger initial load
    LaunchedEffect(productId) {
        viewModel.dispatch(ProductDetailsIntent.Load(productId))
    }

    // Collect one-shot side-effects
    viewModel.collectEffects { effect ->
        when (effect) {
            is ProductDetailsEffect.ShowToast -> showToast(effect.message)
            ProductDetailsEffect.NavigateBack -> onBack()
        }
    }

    ProductDetailsScreen(
        state = state,
        onIntent = viewModel::dispatch,
        onBackClick = { viewModel.dispatch(ProductDetailsIntent.NavigateBack) }
    )
}

4. Write the Stateless Screen

The Screen composable is fully stateless. It relies on the State parameter to render content and emits user actions via the onIntent callback. We use Kolt’s library components here:

// ProductDetailsScreen.kt
@Composable
fun ProductDetailsScreen(
    state: ProductDetailsState,
    onIntent: (ProductDetailsIntent) -> Unit,
    onBackClick: () -> Unit
) {
    AppsPageScaffold(
        title = AppBarTitle.Text("Details"),
        navigationMode = NavigationMode.Back,
        onNavigateBack = onBackClick
    ) {
        // Render system alert banner if message is present
        state.bannerMessage?.let { msg ->
            AppsBanner(
                title = "System Alert",
                message = msg,
                intent = BannerIntent.Warning,
                onAction = { onIntent(ProductDetailsIntent.ClearBanner) }
            )
        }

        // Automate Loading/Error/Success transitions
        AsyncStateBox(
            state = state.asyncState,
            loading = {
                Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    AppsProgressIndicator.CircularIndeterminate()
                }
            },
            error = { error ->
                AppsEmptyState(
                    title = "Failed to load product",
                    description = error.message ?: "An unexpected error occurred.",
                    actionLabel = "Try Again",
                    onAction = { /* dispatch load again if needed */ }
                )
            }
        ) { product ->
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(16.dp)
            ) {
                Text(product.name, style = Kolt.typography.headlineLarge)
                Spacer(Modifier.height(8.dp))
                Text(product.description, style = Kolt.typography.bodyMedium)
                Spacer(Modifier.weight(1f))

                // Action button styled with the primary brand colors
                AppsButton(
                    text = if (state.isFavorite) "Remove Favorite" else "Add to Favorites",
                    buttonStyle = ButtonStyle.primary(),
                    onClick = { onIntent(ProductDetailsIntent.ToggleFavorite) },
                    modifier = Modifier.fillMaxWidth()
                )
            }
        }
    }
}
